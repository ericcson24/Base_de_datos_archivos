
import os
import shutil
import time
from datetime import datetime

EXTENSIONES_VALIDAS = [".txt", ".pdf", ".docx", ".xlsx", ".pptx", ".jpg", ".png", ".mp4", ".mp3"]
EXCLUIR_CARPETAS = ["AppData", "Contacts", "Favorites", "Saved Games", "Searches", "Links", "OneDrive", "NTUSER"]
LOG_FILE = "C:\\NubeDistribuible_v2\\sync_log.txt"

def log(texto):
    timestamp = datetime.now().strftime("[%Y-%m-%d %H:%M:%S]")
    linea = f"{timestamp} {texto}\n"
    print(linea.strip())
    with open(LOG_FILE, "a", encoding="utf-8") as f:
        f.write(linea)

def carpeta_util(carpeta_raiz):
    for raiz, _, archivos in os.walk(carpeta_raiz):
        if any(ex in raiz for ex in EXCLUIR_CARPETAS):
            continue
        for archivo in archivos:
            if any(archivo.lower().endswith(ext) for ext in EXTENSIONES_VALIDAS):
                return True
    return False

def copiar_archivos_utiles(origen, destino):
    for carpeta_raiz, _, archivos in os.walk(origen):
        if any(x in carpeta_raiz for x in EXCLUIR_CARPETAS):
            continue

        if not carpeta_util(carpeta_raiz):
            continue

        ruta_relativa = os.path.relpath(carpeta_raiz, origen)
        carpeta_destino = os.path.join(destino, ruta_relativa)
        os.makedirs(carpeta_destino, exist_ok=True)

        for archivo in archivos:
            if not any(archivo.lower().endswith(ext) for ext in EXTENSIONES_VALIDAS):
                continue

            archivo_origen = os.path.join(carpeta_raiz, archivo)
            archivo_destino = os.path.join(carpeta_destino, archivo)

            try:
                if not os.path.exists(archivo_destino) or os.path.getmtime(archivo_origen) > os.path.getmtime(archivo_destino):
                    shutil.copy2(archivo_origen, archivo_destino)
                    log(f"📁 Copiado: {archivo_origen} -> {archivo_destino}")
            except Exception as e:
                log(f"❌ Error copiando {archivo_origen}: {e}")

def sincronizar_usuario(usuario):
    origen = f"C:\\Users\\{usuario}"
    destino = f"C:\\NubeDistribuible_v2\\Datos\\{usuario}"
    copiar_archivos_utiles(origen, destino)

def obtener_usuarios_activos():
    datos_path = "C:\\NubeDistribuible_v2\\Datos\\usuarios.txt"
    if not os.path.exists(datos_path):
        return []
    with open(datos_path, "r", encoding="utf-8") as f:
        return [linea.split(":")[0].strip() for linea in f if ":" in linea]

if __name__ == "__main__":
    log("🚀 Sincronizador con LOG iniciado (cada 30s)")
    while True:
        usuarios = obtener_usuarios_activos()
        for usuario in usuarios:
            sincronizar_usuario(usuario)
        time.sleep(30)
